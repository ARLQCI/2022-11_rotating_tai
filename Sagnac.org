function V(z; z0=0, E0=0, m=8, depth=2.2MHz)
    return E0 - depth * cos(m*(z - z0))
end

function dV(z; z0=0, E0=0, m=8, depth=2.2MHz)
    return depth * m * sin(m*(z - z0))
end
#+end_src

I don't think we need to simulate from $0$ to $2\pi$ (unless we need that much space to accelerate)

#+begin_src jupyter-julia :session jl :results silent
t_target = 0.02ms
v_target = (2π/8) / t_target
z_target = v_target * t_target

zmin = -π/8
zmax = 3π/8
tlist = collect(range(0, t_target, step=0.05μs))
